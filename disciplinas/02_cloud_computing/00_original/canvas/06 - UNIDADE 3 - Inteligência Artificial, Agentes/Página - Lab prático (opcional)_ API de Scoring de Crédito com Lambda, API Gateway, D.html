Lab prático (opcional): API de Scoring de Crédito com Lambda, API Gateway, DynamoDB e Bedrock

Pessoal,

Durante a aula, construímos na prática uma API serverless de análise de crédito utilizando serviços da AWS.

Neste laboratório, vocês poderão reproduzir a prática no ambiente AWS Academy Data Engineering Sandbox.

A ideia é criar uma função AWS Lambda que calcula uma pontuação de crédito, expor essa função como uma API HTTP com o Amazon API Gateway, chamar a API pelo Postman, gravar cada decisão no Amazon DynamoDB e, por fim, usar um modelo do Amazon Bedrock para explicar a decisão em linguagem simples.

Fluxo da prática

Postman → API Gateway (HTTP API, POST /scoring-credito) → Lambda (scoring-credito) → DynamoDB (decisoes-credito)
                                                                                    → Bedrock (Amazon Nova Micro)

A função passa por três versões ao longo do lab:

  • Versão 1: calcula a pontuação e devolve APROVADO ou REPROVADO;
  • Versão 2: grava cada decisão no DynamoDB;
  • Versão 3: pede ao Bedrock uma explicação da decisão e grava a explicação junto.

A regra de pontuação usa três critérios, somando no máximo 100 pontos. A aprovação exige no mínimo 55 pontos.

CritérioFaixas
Renda mensal (R$ mil)≥ 8: 35 pts | ≥ 5: 25 pts | ≥ 3: 15 pts | abaixo de 3: 5 pts
Score de crédito≥ 800: 40 pts | ≥ 650: 28 pts | ≥ 500: 15 pts | abaixo de 500: 0 pts
Comprometimento (dívida / renda)abaixo de 0,3: 25 pts | abaixo de 0,6: 15 pts | abaixo de 1,0: 5 pts | 1,0 ou mais: 0 pts

import base64
import json

LIMITE_APROVACAO = 55


def calcular_score(renda, score, divida):
    pontos = 0

    # Renda (em milhares de reais)
    if renda >= 8:     pontos += 35
    elif renda >= 5:   pontos += 25
    elif renda >= 3:   pontos += 15
    else:              pontos += 5

    # Score de crédito
    if score >= 800:   pontos += 40
    elif score >= 650: pontos += 28
    elif score >= 500: pontos += 15

    # Comprometimento de renda (dívida / renda)
    comprometimento = divida / renda if renda > 0 else 99

    if comprometimento < 0.3:   pontos += 25
    elif comprometimento < 0.6: pontos += 15
    elif comprometimento < 1.0: pontos += 5

    return pontos  # máximo: 100


def resposta(status, corpo):
    return {
        "statusCode": status,
        "headers": {"Content-Type": "application/json"},
        "body": json.dumps(corpo, ensure_ascii=False),
    }


def lambda_handler(event, context):
    try:
        body = event.get("body", event)
        if event.get("isBase64Encoded"):
            body = base64.b64decode(body).decode("utf-8")
        if isinstance(body, str):
            body = json.loads(body)

        renda  = float(body["renda_mensal"])
        score  = float(body["score_credito"])
        divida = float(body["divida_atual"])

        pontos   = calcular_score(renda, score, divida)
        aprovado = pontos >= LIMITE_APROVACAO

        return resposta(200, {
            "aprovado": aprovado,
            "decisao": "APROVADO" if aprovado else "REPROVADO",
            "pontuacao": pontos,
            "limite_min": LIMITE_APROVACAO,
            "dados": {"renda_mensal": renda, "score_credito": score,
                      "divida_atual": divida},
        })

    except KeyError as e:
        return resposta(400, {"erro": f"Campo ausente: {e}"})
    except (ValueError, TypeError):
        return resposta(400, {"erro": "Payload inválido: envie JSON com valores numéricos"})
    except Exception as e:
        return resposta(500, {"erro": str(e)})

import base64
import json
import os
import uuid
from datetime import datetime, timezone
from decimal import Decimal

import boto3

LIMITE_APROVACAO = 55
TABELA = os.environ.get("TABELA", "decisoes-credito")

# Cliente criado fora do handler: reaproveitado no warm start
tabela = boto3.resource("dynamodb").Table(TABELA)


def calcular_score(renda, score, divida):
    pontos = 0

    # Renda (em milhares de reais)
    if renda >= 8:     pontos += 35
    elif renda >= 5:   pontos += 25
    elif renda >= 3:   pontos += 15
    else:              pontos += 5

    # Score de crédito
    if score >= 800:   pontos += 40
    elif score >= 650: pontos += 28
    elif score >= 500: pontos += 15

    # Comprometimento de renda (dívida / renda)
    comprometimento = divida / renda if renda > 0 else 99

    if comprometimento < 0.3:   pontos += 25
    elif comprometimento < 0.6: pontos += 15
    elif comprometimento < 1.0: pontos += 5

    return pontos  # máximo: 100


def resposta(status, corpo):
    return {
        "statusCode": status,
        "headers": {"Content-Type": "application/json"},
        "body": json.dumps(corpo, ensure_ascii=False),
    }


def lambda_handler(event, context):
    try:
        body = event.get("body", event)
        if event.get("isBase64Encoded"):
            body = base64.b64decode(body).decode("utf-8")
        if isinstance(body, str):
            body = json.loads(body)

        renda  = float(body["renda_mensal"])
        score  = float(body["score_credito"])
        divida = float(body["divida_atual"])

        pontos   = calcular_score(renda, score, divida)
        aprovado = pontos >= LIMITE_APROVACAO
        decisao  = "APROVADO" if aprovado else "REPROVADO"

            "id_decisao": id_decisao,
            "data_hora": datetime.now(timezone.utc).isoformat(),
            "renda_mensal": Decimal(str(renda)),
            "score_credito": Decimal(str(score)),
            "divida_atual": Decimal(str(divida)),
            "pontuacao": pontos,
            "decisao": decisao,

        return resposta(200, {
            "id_decisao": id_decisao,
            "aprovado": aprovado,
            "decisao": decisao,
            "pontuacao": pontos,

import base64
import json
import os
import uuid
from datetime import datetime, timezone
from decimal import Decimal

import boto3

LIMITE_APROVACAO = 55
TABELA = os.environ.get("TABELA", "decisoes-credito")
MODEL_ID = os.environ.get("MODEL_ID", "amazon.nova-micro-v1:0")

# Clientes criados fora do handler: reaproveitados no warm start
tabela = boto3.resource("dynamodb").Table(TABELA)
bedrock = boto3.client("bedrock-runtime")


def calcular_score(renda, score, divida):
    pontos = 0

    # Renda (em milhares de reais)
    if renda >= 8:     pontos += 35
    elif renda >= 5:   pontos += 25
    elif renda >= 3:   pontos += 15
    else:              pontos += 5

    # Score de crédito
    if score >= 800:   pontos += 40
    elif score >= 650: pontos += 28
    elif score >= 500: pontos += 15

    # Comprometimento de renda (dívida / renda)
    comprometimento = divida / renda if renda > 0 else 99

    if comprometimento < 0.3:   pontos += 25
    elif comprometimento < 0.6: pontos += 15
    elif comprometimento < 1.0: pontos += 5

    return pontos  # máximo: 100


def gerar_explicacao(renda, score, divida, pontos, decisao):
    # A LLM não decide nada: só explica, em linguagem simples, a decisão já tomada
    comprometimento = divida / renda if renda > 0 else 0
    prompt = (
        "Você é um analista de crédito. Explique ao cliente, em no máximo 3 frases, "
        "em português e sem jargão, o motivo da decisão abaixo. "
        "Use apenas os dados informados e não invente critérios.\n\n"
        f"Renda mensal: R$ {renda:.1f} mil\n"
        f"Score de crédito: {score:.0f}\n"
        f"Dívida atual: R$ {divida:.1f} mil\n"
        f"Comprometimento da renda (dívida/renda): {comprometimento:.2f}\n"
        f"Pontuação: {pontos} de 100 (mínimo para aprovar: {LIMITE_APROVACAO})\n"
        f"Decisão: {decisao}"
    )
    r = bedrock.converse(
        modelId=MODEL_ID,
        messages=[{"role": "user", "content": [{"text": prompt}]}],
        inferenceConfig={"maxTokens": 200, "temperature": 0.2},
    )
    return r["output"]["message"]["content"][0]["text"]


def resposta(status, corpo):
    return {
        "statusCode": status,
        "headers": {"Content-Type": "application/json"},
        "body": json.dumps(corpo, ensure_ascii=False),
    }


def lambda_handler(event, context):
    try:
        body = event.get("body", event)
        if event.get("isBase64Encoded"):
            body = base64.b64decode(body).decode("utf-8")
        if isinstance(body, str):
            body = json.loads(body)

        renda  = float(body["renda_mensal"])
        score  = float(body["score_credito"])
        divida = float(body["divida_atual"])

        pontos   = calcular_score(renda, score, divida)
        aprovado = pontos >= LIMITE_APROVACAO
        decisao  = "APROVADO" if aprovado else "REPROVADO"

        # Se o Bedrock falhar, a decisão continua saindo
        try:
            explicacao = gerar_explicacao(renda, score, divida, pontos, decisao)
        except Exception as e:
            explicacao = f"Explicação indisponível: {e}"

        # Grava a decisão e a explicação no DynamoDB (DynamoDB não aceita float: usa Decimal)
        id_decisao = str(uuid.uuid4())
        tabela.put_item(Item={
            "id_decisao": id_decisao,
            "data_hora": datetime.now(timezone.utc).isoformat(),
            "renda_mensal": Decimal(str(renda)),
            "score_credito": Decimal(str(score)),
            "divida_atual": Decimal(str(divida)),
            "pontuacao": pontos,
            "decisao": decisao,
            "explicacao": explicacao,
            "modelo": MODEL_ID,
        })

        return resposta(200, {
            "id_decisao": id_decisao,
            "aprovado": aprovado,
            "decisao": decisao,
            "pontuacao": pontos,
            "limite_min": LIMITE_APROVACAO,
            "explicacao": explicacao,
            "dados": {"renda_mensal": renda, "score_credito": score,
                      "divida_atual": divida},
        })

    except KeyError as e:
        return resposta(400, {"erro": f"Campo ausente: {e}"})
    except (ValueError, TypeError):
        return resposta(400, {"erro": "Payload inválido: envie JSON com valores numéricos"})
    except Exception as e:
        return resposta(500, {"erro": str(e)})

O que você deve aprender com a prática

Ao finalizar o laboratório, você deverá conseguir explicar:

  • Qual é a função do AWS Lambda e o que é o evento que ele recebe;
  • Por que a resposta no teste do console vem com statusCode e body, e no Postman vem só o JSON;
  • Qual é a função do API Gateway na frente da Lambda;
  • Por que a rota aceita apenas POST, e o que acontece com uma chamada GET;
  • O que são o estágio $default e a implantação automática;
  • Por que uma reprovação volta com status 200, e quando a API devolve 400 ou 500;
  • Por que o código usa Decimal para gravar números no DynamoDB;
  • Por que os clientes do DynamoDB e do Bedrock são criados fora do handler;
  • Por que a permissão do Bedrock precisa estar na role de execução da Lambda;
  • Por que, no código, a decisão vem da regra e o modelo apenas explica;
  • Por que a chamada ao Bedrock fica dentro de um try separado;
  • Como diferentes serviços AWS podem ser combinados para construir uma API serverless com IA generativa.

Postman → API Gateway → Lambda → regra de pontuação → Bedrock (explicação) → DynamoDB → resposta JSON
